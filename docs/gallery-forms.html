	if trim(oEdit.Text) != ""
		oList.Items.Add(oEdit.Text)

	if oList.SelectedIndex >= 0
		oList.Items.RemoveAt(oList.SelectedIndex)
	ok

	oList.SelectedIndex = 1
	dnAssert("selected text",oSelected.Text,"Selected: green")
	removeItem()
